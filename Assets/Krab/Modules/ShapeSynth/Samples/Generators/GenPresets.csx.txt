// Generates presets by random binding, keeps only the ones that pass numeric checks, and saves them as "Gen/…".
// One Shape node each (Layout copies allowed), white, shared Values only. Deterministic (fixed seed); names that
// already exist are skipped, so it can be re-run.
var repository = new PresetRepository();
var values = repository.LoadValues();
var existing = new HashSet<string>(repository.List(out _).Select(p => p.Name));
var rng = new System.Random(20261126);
T Pick<T>(IReadOnlyList<T> items) => items[rng.Next(items.Count)];
double Range(double a, double b) => a + rng.NextDouble() * (b - a);

var envelopes = values.Where(v => v.Name.StartsWith("Eased/") || v.Name.StartsWith("Motion/") || v.Name == "ADSR/Fast")
    .Where(v => v.Components.All(c => c.Kind == ModulatorKind.Adsr)).ToList();
var motions = values.Where(v => v.Name.StartsWith("Eased/") || v.Name.StartsWith("Motion/") || v.Name.StartsWith("LFO/")).ToList();
var canvas = new ReferenceContext(1920, 1080);
var signatures = new List<double[]>();
var log = new List<string>();
int saved = 0, tried = 0;
var reasons = new Dictionary<string, int>();
void Reject(string why) { reasons[why] = reasons.TryGetValue(why, out var n) ? n + 1 : 1; }

string[] targets = { "PositionX", "PositionY", "Rotation", "Scale", "ScaleX", "ScaleY", "Size", "Spread" };
while (saved < 100 && tried < 4000)
{
    tried++;
    var root = PresetNode.Create(PresetNodeKind.Group); root.Name = "Root Group";
    var s = PresetNode.Create(rng.NextDouble() < 0.5 ? PresetNodeKind.Rectangle : PresetNodeKind.Circle);
    s.Name = s.Kind.ToString();
    root.Children.Add(s);
    var document = new PresetDocument { Id = PresetDocument.NewId(), Name = "draft", RootGroup = root, Values = values };

    // Shape and placement.
    var big = rng.NextDouble() < 0.3;
    s.Width.Reference = ReferenceSource.MinDimension; s.Height.Reference = ReferenceSource.MinDimension;
    s.Width.Base = big ? Range(0.2, 0.5) : Range(0.02, 0.12);
    s.Height.Base = s.Kind == PresetNodeKind.Rectangle && rng.NextDouble() < 0.5 ? s.Width.Base * Pick(new[] { 0.1, 0.25, 4.0, 8.0 }) : s.Width.Base;
    if (s.Kind == PresetNodeKind.Rectangle && rng.NextDouble() < 0.35) s.Anchor = Pick((PresetAnchor[])Enum.GetValues(typeof(PresetAnchor)));
    if (rng.NextDouble() < 0.2) { s.Fill = false; s.Stroke = true; s.StrokeWidth.Reference = ReferenceSource.MinDimension; s.StrokeWidth.Base = 0.005; }

    var layouts = new[] { PresetLayoutKind.None, PresetLayoutKind.Row, PresetLayoutKind.Column, PresetLayoutKind.Grid, PresetLayoutKind.Circle };
    var layout = big ? PresetLayoutKind.None : Pick(layouts);
    s.Layout.Kind = layout;
    if (layout != PresetLayoutKind.None)
    {
        s.Layout.Count = layout == PresetLayoutKind.Grid ? Pick(new[] { 9, 16, 25 }) : Pick(new[] { 3, 5, 8, 12, 16 });
        s.Layout.Columns = (int)Math.Round(Math.Sqrt(s.Layout.Count));
        s.Layout.SpacingX.Reference = ReferenceSource.MinDimension; s.Layout.SpacingY.Reference = ReferenceSource.MinDimension;
        s.Layout.SpacingX.Base = Range(0.08, 0.9 / Math.Max(1, layout == PresetLayoutKind.Grid ? s.Layout.Columns : s.Layout.Count));
        s.Layout.SpacingY.Base = s.Layout.SpacingX.Base;
        s.Layout.Radius.Reference = ReferenceSource.MinDimension; s.Layout.Radius.Base = Range(0.15, 0.4);
        s.Layout.FaceOutward = rng.NextDouble() < 0.5;
        if (rng.NextDouble() < 0.6)
        {
            s.Layout.Delay = Pick(new[] { 0.03, 0.06, 0.125, 0.25 });
            s.Layout.DelayOrder = Pick((PresetDelayOrder[])Enum.GetValues(typeof(PresetDelayOrder)));
        }
    }
    else if (!big && rng.NextDouble() < 0.5)
    {
        // A single small shape: one of a few fixed places.
        s.PositionX.Reference = ReferenceSource.CanvasWidth; s.PositionX.Random = 0.4; s.PositionX.RandomSteps = Pick(new[] { 3, 5 });
    }

    // Envelope on Opacity (the Voice's lifetime) and one or two motions.
    s.Opacity.Base = 0; s.Opacity.Bind(Pick(envelopes).Id, 1);
    var chosen = targets.OrderBy(_ => rng.Next()).Take(rng.Next(1, 3)).ToList();
    if (layout == PresetLayoutKind.None) chosen.Remove("Spread");
    foreach (var target in chosen)
    {
        var v = Pick(motions);
        switch (target)
        {
            case "PositionX": s.PositionX.Reference = ReferenceSource.CanvasWidth; s.PositionX.Base -= 0.15; s.PositionX.Bind(v.Id, 0.3); break;
            case "PositionY": s.PositionY.Reference = ReferenceSource.CanvasHeight; s.PositionY.Base = -0.15; s.PositionY.Bind(v.Id, 0.3); break;
            case "Rotation": s.Rotation.Bind(v.Id, Pick(new[] { 45.0, 90, 180 })); if (s.Layout.Kind != PresetLayoutKind.None && rng.NextDouble() < 0.5) s.Rotation.Flip = PresetFlip.Alternate; break;
            case "Scale": s.ScaleX.Base = 0; s.ScaleY.Base = 0; s.ScaleX.Bind(v.Id, 1); s.ScaleY.Bind(v.Id, 1); break;
            case "ScaleX": s.ScaleX.Base = 0; s.ScaleX.Bind(v.Id, 1); break;
            case "ScaleY": s.ScaleY.Base = 0; s.ScaleY.Bind(v.Id, 1); break;
            case "Size": s.Width.Bind(v.Id, s.Width.Base); if (s.Kind == PresetNodeKind.Rectangle) s.Height.Bind(v.Id, s.Height.Base); break;
            case "Spread":
                if (layout == PresetLayoutKind.Circle) { s.Layout.Radius.Base *= 0.3; s.Layout.Radius.Bind(v.Id, 0.3); }
                else { s.Layout.SpacingX.Base *= 0.3; s.Layout.SpacingY.Base *= 0.3; s.Layout.SpacingX.Bind(v.Id, s.Layout.SpacingX.Base * 2); s.Layout.SpacingY.Bind(v.Id, s.Layout.SpacingY.Base * 2); }
                break;
        }
    }

    // Numeric checks.
    PresetCompiler.Compile(document);
    var components = PresetCompiler.Components(document).ToArray();
    var lifetime = components.Where(c => c.Kind == ModulatorKind.Adsr).Max(c => c.Attack + c.Decay + c.SustainTime + c.Release)
        + PresetCopyPattern.MaxDelayBeats(s.Layout);
    if (lifetime < 0.5 || lifetime > 4) { Reject("length"); continue; }

    var group = PresetCompiler.CompileGroup(document);
    var time = new VoiceTime(0, 60);
    var signals = components.Select(c => PresetCompiler.ModulatorDefinition(c).Create(time)).ToArray();
    var evaluator = new HierarchyEvaluator(group, signals, 7);
    const int samples = 24;
    var signature = new double[samples * 3];
    double inside = 0, total = 0, visible = 0, motion = 0;
    double[] previous = null;
    for (var i = 0; i < samples; i++)
    {
        var items = evaluator.Evaluate(lifetime * (i + 0.5) / samples, canvas);
        double cx = 0, cy = 0, size = 0, alpha = 0;
        foreach (var item in items)
        {
            var c = item.WorldTransform.TransformPoint(new Point2(0, 0));
            var e = item.WorldTransform.TransformPoint(new Point2(Math.Max(item.Width, 1) / 2, Math.Max(item.Height, 1) / 2));
            var r = Math.Sqrt((e.X - c.X) * (e.X - c.X) + (e.Y - c.Y) * (e.Y - c.Y));
            total++;
            if (Math.Abs(c.X) <= 960 && Math.Abs(c.Y) <= 540) inside++;
            cx += c.X; cy += c.Y; size += r; alpha += item.Opacity;
        }

        var n = Math.Max(1, items.Count);
        var frame = new[] { cx / n / 960, cy / n / 540, size / n / 540 };
        if (alpha / n > 0.2) visible++;
        if (previous != null) motion += Math.Abs(frame[0] - previous[0]) + Math.Abs(frame[1] - previous[1]) + Math.Abs(frame[2] - previous[2]);
        previous = frame;
        signature[i * 3] = frame[0]; signature[i * 3 + 1] = frame[1]; signature[i * 3 + 2] = frame[2];
    }

    if (total == 0 || inside / total < 0.85) { Reject("offscreen"); continue; }
    if (visible < samples * 0.3) { Reject("invisible"); continue; }
    if (motion < 0.08) { Reject("still"); continue; }
    if (motion > 6) { Reject("jitter"); continue; }
    if (signatures.Any(other => other.Zip(signature, (a, b) => Math.Abs(a - b)).Average() < 0.02)) { Reject("similar"); continue; }

    signatures.Add(signature);
    saved++;
    var name = $"Gen/{saved:000} {s.Kind} {layout} {string.Join("+", chosen)}";
    if (existing.Contains(name)) continue;
    document.Name = name;
    repository.Save(document);
}

return $"saved {saved} of {tried} tried; rejected " + string.Join(", ", reasons.Select(kv => kv.Key + " " + kv.Value));
